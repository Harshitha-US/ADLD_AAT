`timescale 1ns/1ps
module tb_alu4;
reg [3:0] A, B;
reg [2:0] sel;
wire [3:0] Y;
wire C;
alu4 dut(.A(A), .B(B), .sel(sel), .Y(Y), .C(C));
initial begin
    A=4'd9; B=4'd6;
    sel=3'b000; #10; $display("AND: A=%d B=%d Y=%d C=%b",A,B,Y,C);
    sel=3'b001; #10; $display("OR : Y=%d",Y);
    sel=3'b010; #10; $display("XOR: Y=%d",Y);
    sel=3'b011; #10; $display("ADD: Y=%d C=%b",Y,C);
    sel=3'b100; #10; $display("SUB: Y=%d Borrow=%b",Y,C);
    sel=3'b101; #10; $display("PASS A: Y=%d",Y);
    sel=3'b110; #10; $display("PASS B: Y=%d",Y);
    sel=3'b111; #10; $display("ZERO: Y=%d",Y);
    $finish;
end
endmodule
